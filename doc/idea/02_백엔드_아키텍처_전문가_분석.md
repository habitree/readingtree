### 3.1 복합 인덱스 최적화

#### 현재 문제
- 단일 컬럼 인덱스만 존재
- 자주 사용되는 쿼리 패턴에 대한 복합 인덱스 부재
- 대용량 데이터 시 성능 저하 예상

#### 개선 방안: 복합 인덱스 추가

```sql
-- doc/database/migration-001_indexes.sql

-- 1. user_books 테이블: 사용자별 상태 필터링
-- 자주 사용되는 쿼리: user_id + status + created_at
CREATE INDEX idx_user_books_user_status_created
    ON user_books(user_id, status, created_at DESC);

-- 2. notes 테이블: 사용자별 책 기록 조회
-- 자주 사용되는 쿼리: user_id + book_id + created_at
CREATE INDEX idx_notes_user_book_created
    ON notes(user_id, book_id, created_at DESC);

-- 3. notes 테이블: 타임라인 조회 (공개 기록)
-- 자주 사용되는 쿼리: is_public + created_at
CREATE INDEX idx_notes_public_created
    ON notes(is_public, created_at DESC)
    WHERE is_public = TRUE;

-- 4. group_members 테이블: 그룹별 승인된 멤버 조회
CREATE INDEX idx_group_members_group_status
    ON group_members(group_id, status)
    WHERE status = 'approved';

-- 5. books 테이블: ISBN 조회 최적화
CREATE INDEX idx_books_isbn_partial
    ON books(isbn)
    WHERE isbn IS NOT NULL;

-- 6. 통계 쿼리용 부분 인덱스
-- 올해 완독 책 수 조회 최적화
CREATE INDEX idx_user_books_completed_year
    ON user_books(user_id, completed_at)
    WHERE status = 'completed'
    AND completed_at >= date_trunc('year', CURRENT_DATE);
```

### 3.2 Upstash Redis 캐싱/Rate Limiting

#### 현재 문제
- API 응답 캐싱 없음
- Rate Limiting 미구현
- 외부 API 호출 최적화 부재

#### 개선 방안: Upstash Redis 도입

```typescript
// lib/redis.ts
import { Redis } from '@upstash/redis';

export const redis = new Redis({
  url: process.env.UPSTASH_REDIS_REST_URL!,
  token: process.env.UPSTASH_REDIS_REST_TOKEN!,
});

// 캐시 유틸리티
export const cache = {
  async get<T>(key: string): Promise<T | null> {
    return redis.get<T>(key);
  },

  async set<T>(key: string, value: T, ttlSeconds?: number): Promise<void> {
    if (ttlSeconds) {
      await redis.setex(key, ttlSeconds, value);
    } else {
      await redis.set(key, value);
    }
  },

  async del(key: string): Promise<void> {
    await redis.del(key);
  },

  // 패턴 기반 삭제 (캐시 무효화)
  async invalidate(pattern: string): Promise<void> {
    const keys = await redis.keys(pattern);
    if (keys.length > 0) {
      await redis.del(...keys);
    }
  },
};

// 캐시 키 생성 헬퍼
export const cacheKeys = {
  userBooks: (userId: string) => `user:${userId}:books`,
  userStats: (userId: string) => `user:${userId}:stats`,
  bookSearch: (query: string) => `search:book:${query}`,
  dashboardData: (userId: string) => `dashboard:${userId}`,
};
```

#### Rate Limiting 구현

```typescript
// lib/rate-limit.ts
import { Ratelimit } from '@upstash/ratelimit';
import { redis } from './redis';

// API별 Rate Limiter 설정
export const rateLimiters = {
  // 일반 API: 분당 60회
  default: new Ratelimit({
    redis,
    limiter: Ratelimit.slidingWindow(60, '1 m'),
    analytics: true,
    prefix: 'ratelimit:default',
  }),

  // 검색 API: 분당 30회
  search: new Ratelimit({
    redis,
    limiter: Ratelimit.slidingWindow(30, '1 m'),
    analytics: true,
    prefix: 'ratelimit:search',
  }),

  // OCR API: 시간당 10회
  ocr: new Ratelimit({
    redis,
    limiter: Ratelimit.slidingWindow(10, '1 h'),
    analytics: true,
    prefix: 'ratelimit:ocr',
  }),

  // 인증 API: 분당 5회 (브루트포스 방지)
  auth: new Ratelimit({
    redis,
    limiter: Ratelimit.slidingWindow(5, '1 m'),
    analytics: true,
    prefix: 'ratelimit:auth',
  }),
};

// Rate Limit 체크 함수
export async function checkRateLimit(
  limiter: Ratelimit,
  identifier: string
): Promise<{ success: boolean; limit: number; remaining: number; reset: number }> {
  const result = await limiter.limit(identifier);
  return {
    success: result.success,
    limit: result.limit,
    remaining: result.remaining,
    reset: result.reset,
  };
}
```

#### 미들웨어에서 Rate Limit 적용

```typescript
// middleware.ts
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';
import { rateLimiters, checkRateLimit } from '@/lib/rate-limit';

export async function middleware(req: NextRequest) {
  // API 라우트에만 Rate Limit 적용
  if (req.nextUrl.pathname.startsWith('/api/')) {
    const ip = req.ip ?? req.headers.get('x-forwarded-for') ?? '127.0.0.1';
    const path = req.nextUrl.pathname;

    // API 유형에 따른 Rate Limiter 선택
    let limiter = rateLimiters.default;
    if (path.includes('/search')) {
      limiter = rateLimiters.search;
    } else if (path.includes('/ocr')) {
      limiter = rateLimiters.ocr;
    }

    const result = await checkRateLimit(limiter, ip);

    if (!result.success) {
      return NextResponse.json(
        {
          error: 'Too many requests',
          retryAfter: Math.ceil((result.reset - Date.now()) / 1000),
        },
        {
          status: 429,
          headers: {
            'X-RateLimit-Limit': result.limit.toString(),
            'X-RateLimit-Remaining': result.remaining.toString(),
            'X-RateLimit-Reset': result.reset.toString(),
            'Retry-After': Math.ceil((result.reset - Date.now()) / 1000).toString(),
          },
        }
      );
    }
  }

  return NextResponse.next();
}
```

#### 캐싱 적용 예시

```typescript
// lib/api/books.ts
import { cache, cacheKeys } from '@/lib/redis';
import { createServerClient } from '@/lib/supabase/server';

export async function getUserBooks(userId: string) {
  const cacheKey = cacheKeys.userBooks(userId);

  // 캐시 확인
  const cached = await cache.get(cacheKey);
  if (cached) {
    return cached;
  }

  // DB 조회
  const supabase = createServerClient();
  const { data, error } = await supabase
    .from('user_books')
    .select(`
      *,
      book:books(*)
    `)
    .eq('user_id', userId)
    .order('created_at', { ascending: false });

  if (error) throw error;

  // 캐시 저장 (5분 TTL)
  await cache.set(cacheKey, data, 300);

  return data;
}

// 책 추가 시 캐시 무효화
export async function addBook(userId: string, bookData: any) {
  const supabase = createServerClient();

  // DB에 추가
  const { data, error } = await supabase
    .from('user_books')
    .insert({
      user_id: userId,
      book_id: bookData.bookId,
      status: 'reading',
    })
    .select()
    .single();

  if (error) throw error;

  // 캐시 무효화
  await cache.del(cacheKeys.userBooks(userId));
  await cache.del(cacheKeys.userStats(userId));
  await cache.del(cacheKeys.dashboardData(userId));

  return data;
}
```

#### OCR 처리 함수

```typescript
// lib/inngest/functions/ocr.ts
import { inngest } from '../client';
import { extractTextFromImage } from '@/lib/api/gemini';
import { createAdminClient } from '@/lib/supabase/admin';

export const processOcr = inngest.createFunction(
  {
    id: 'process-ocr',
    name: 'Process OCR',
    retries: 3,
    throttle: {
      limit: 10, // 동시 실행 10개 제한
      period: '1m',
    },
  },
  { event: 'ocr/process' },
  async ({ event, step }) => {
    const { noteId, imageUrl, userId } = event.data;

    // Step 1: OCR 처리
    const extractedText = await step.run('extract-text', async () => {
      return extractTextFromImage(imageUrl);
    });

    // Step 2: 텍스트 정제
    const cleanedText = await step.run('clean-text', async () => {
      // 줄바꿈 정규화, 불필요한 공백 제거 등
      return extractedText
        .replace(/\r\n/g, '\n')
        .replace(/\n{3,}/g, '\n\n')
        .trim();
    });

    // Step 3: DB 업데이트
    await step.run('update-note', async () => {
      const supabase = createAdminClient();

      const { error } = await supabase
        .from('notes')
        .update({
          content: cleanedText,
          ocr_processed: true,
          ocr_processed_at: new Date().toISOString(),
        })
        .eq('id', noteId);

      if (error) throw error;
    });

    // Step 4: 사용자 알림
    await step.run('notify-user', async () => {
      // Realtime 또는 푸시 알림으로 완료 통지
      const supabase = createAdminClient();
      await supabase
        .from('notifications')
        .insert({
          user_id: userId,
          type: 'ocr_complete',
          title: 'OCR 처리 완료',
          message: '이미지에서 텍스트 추출이 완료되었습니다.',
          data: { noteId },
        });
    });

    return { success: true, noteId, textLength: cleanedText.length };
  }
);
```

#### AI 요약 생성 함수

```typescript
// lib/inngest/functions/ai-summary.ts
import { inngest } from '../client';
import { generateSummary } from '@/lib/api/gemini';
import { createAdminClient } from '@/lib/supabase/admin';

export const generateBookSummary = inngest.createFunction(
  {
    id: 'generate-book-summary',
    name: 'Generate Book Summary',
    retries: 2,
  },
  { event: 'ai/generate-summary' },
  async ({ event, step }) => {
    const { bookId, userId, noteIds } = event.data;
    const supabase = createAdminClient();

    // Step 1: 기록 내용 수집
    const notes = await step.run('fetch-notes', async () => {
      const { data, error } = await supabase
        .from('notes')
        .select('content, type, page_number')
        .in('id', noteIds)
        .order('page_number', { ascending: true });

      if (error) throw error;
      return data;
    });

    // Step 2: AI 요약 생성
    const summary = await step.run('generate-summary', async () => {
      const notesText = notes
        .map((n) => `[${n.type}${n.page_number ? ` p.${n.page_number}` : ''}] ${n.content}`)
        .join('\n\n');

      return generateSummary(notesText);
    });

    // Step 3: 요약 저장
    await step.run('save-summary', async () => {
      const { error } = await supabase
        .from('book_summaries')
        .upsert({
          book_id: bookId,
          user_id: userId,
          summary,
          generated_at: new Date().toISOString(),
        });

      if (error) throw error;
    });

    return { success: true, bookId, summaryLength: summary.length };
  }
);
```

#### API 엔드포인트에서 이벤트 발송

```typescript
// app/api/ocr/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { inngest } from '@/lib/inngest/client';
import { createServerClient } from '@/lib/supabase/server';

export async function POST(request: NextRequest) {
  const supabase = createServerClient();

  // 인증 확인
  const { data: { user }, error: authError } = await supabase.auth.getUser();
  if (authError || !user) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  }

  const { noteId, imageUrl } = await request.json();

  // Note 소유권 확인
  const { data: note, error: noteError } = await supabase
    .from('notes')
    .select('id')
    .eq('id', noteId)
    .eq('user_id', user.id)
    .single();

  if (noteError || !note) {
    return NextResponse.json({ error: 'Note not found' }, { status: 404 });
  }

  // Inngest 이벤트 발송 (비동기 처리 시작)
  await inngest.send({
    name: 'ocr/process',
    data: {
      noteId,
      imageUrl,
      userId: user.id,
    },
  });

  // 즉시 응답 (처리 중 상태)
  return NextResponse.json({
    success: true,
    message: 'OCR 처리가 시작되었습니다. 완료 시 알림을 받으실 수 있습니다.',
    noteId,
  });
}
```

```sql
-- 1. 사용자 대시보드 통계 조회 함수
CREATE OR REPLACE FUNCTION get_user_dashboard_stats(p_user_id UUID)
RETURNS TABLE (
    total_books BIGINT,
    reading_books BIGINT,
    completed_books BIGINT,
    total_notes BIGINT,

-- 2. 책 추가 + 사용자 서재 등록 트랜잭션
CREATE OR REPLACE FUNCTION add_book_to_library(
    p_user_id UUID,
    p_isbn VARCHAR,
    p_title VARCHAR,
    p_author VARCHAR,
    p_publisher VARCHAR,
    p_published_date DATE,
    p_cover_image_url TEXT
)
RETURNS UUID AS $$
DECLARE
    v_book_id UUID;
    v_user_book_id UUID;
BEGIN
    -- 기존 책 확인 (ISBN이 있는 경우)
    IF p_isbn IS NOT NULL THEN
        SELECT id INTO v_book_id
        FROM books
        WHERE isbn = p_isbn
        LIMIT 1;
    END IF;

    -- 책이 없으면 새로 생성
    IF v_book_id IS NULL THEN
        INSERT INTO books (isbn, title, author, publisher, published_date, cover_image_url)
        VALUES (p_isbn, p_title, p_author, p_publisher, p_published_date, p_cover_image_url)
        RETURNING id INTO v_book_id;
    END IF;

    -- 사용자 서재에 추가 (이미 있으면 무시)
    INSERT INTO user_books (user_id, book_id, status, started_at)
    VALUES (p_user_id, v_book_id, 'reading', NOW())
    ON CONFLICT (user_id, book_id) DO NOTHING
    RETURNING id INTO v_user_book_id;

    RETURN v_book_id;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 3. 스트릭 업데이트 함수
CREATE OR REPLACE FUNCTION update_user_streak(p_user_id UUID)
RETURNS void AS $$
DECLARE
    v_last_activity_date DATE;
    v_current_streak INTEGER;
    v_longest_streak INTEGER;
BEGIN
    -- 현재 스트릭 정보 조회
    SELECT last_activity_date, streak, longest_streak
    INTO v_last_activity_date, v_current_streak, v_longest_streak
    FROM user_streaks
    WHERE user_id = p_user_id;

    -- 첫 활동인 경우
    IF v_last_activity_date IS NULL THEN
        INSERT INTO user_streaks (user_id, streak, longest_streak, last_activity_date)
        VALUES (p_user_id, 1, 1, CURRENT_DATE);
        RETURN;
    END IF;

    -- 이미 오늘 활동한 경우
    IF v_last_activity_date = CURRENT_DATE THEN
        RETURN;
    END IF;

    -- 어제 활동한 경우 (스트릭 유지)
    IF v_last_activity_date = CURRENT_DATE - INTERVAL '1 day' THEN
        v_current_streak := v_current_streak + 1;
        IF v_current_streak > v_longest_streak THEN
            v_longest_streak := v_current_streak;
        END IF;
    ELSE
        -- 스트릭 끊김
        v_current_streak := 1;
    END IF;

    UPDATE user_streaks
    SET streak = v_current_streak,
        longest_streak = v_longest_streak,
        last_activity_date = CURRENT_DATE
    WHERE user_id = p_user_id;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 4. 기록 작성 시 스트릭 자동 업데이트 트리거
CREATE OR REPLACE FUNCTION trigger_update_streak_on_note()
RETURNS TRIGGER AS $$
BEGIN
    PERFORM update_user_streak(NEW.user_id);
    RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE TRIGGER on_note_created
    AFTER INSERT ON notes
    FOR EACH ROW
    EXECUTE FUNCTION trigger_update_streak_on_note();
```

```sql
-- doc/database/migration-002_audit.sql

-- Audit 로그 테이블
CREATE TABLE audit_logs (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    table_name VARCHAR(100) NOT NULL,
    record_id UUID NOT NULL,
    action VARCHAR(20) NOT NULL, -- INSERT, UPDATE, DELETE
    old_data JSONB,
    new_data JSONB,
    changed_fields TEXT[],
    user_id UUID REFERENCES auth.users(id),
    ip_address INET,
    user_agent TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 인덱스
CREATE INDEX idx_audit_logs_table_record ON audit_logs(table_name, record_id);
CREATE INDEX idx_audit_logs_user ON audit_logs(user_id);
CREATE INDEX idx_audit_logs_created ON audit_logs(created_at DESC);
CREATE INDEX idx_audit_logs_action ON audit_logs(action);

-- 파티셔닝 (월별) - 대용량 로그 관리
-- 프로덕션 환경에서는 파티션 테이블로 전환 권장

-- 감사 로그 생성 함수
CREATE OR REPLACE FUNCTION audit_log_changes()
RETURNS TRIGGER AS $$
DECLARE
    v_old_data JSONB;
    v_new_data JSONB;
    v_changed_fields TEXT[];
    v_user_id UUID;
    v_key TEXT;
BEGIN
    -- 현재 사용자 ID 가져오기 (Supabase auth.uid())
    v_user_id := auth.uid();

    IF TG_OP = 'INSERT' THEN
        v_new_data := to_jsonb(NEW);
        v_old_data := NULL;
        v_changed_fields := ARRAY(SELECT jsonb_object_keys(v_new_data));

    ELSIF TG_OP = 'UPDATE' THEN
        v_old_data := to_jsonb(OLD);
        v_new_data := to_jsonb(NEW);

        -- 변경된 필드만 추출
        v_changed_fields := ARRAY(
            SELECT key
            FROM jsonb_each(v_new_data) AS n(key, value)
            WHERE v_old_data->key IS DISTINCT FROM n.value
        );

        -- 변경 사항이 없으면 로그 생성 안 함
        IF array_length(v_changed_fields, 1) IS NULL THEN
            RETURN NEW;
        END IF;

    ELSIF TG_OP = 'DELETE' THEN
        v_old_data := to_jsonb(OLD);
        v_new_data := NULL;
        v_changed_fields := NULL;
    END IF;

    -- 감사 로그 삽입
    INSERT INTO audit_logs (
        table_name,
        record_id,
        action,
        old_data,
        new_data,
        changed_fields,
        user_id
    ) VALUES (
        TG_TABLE_NAME,
        COALESCE(NEW.id, OLD.id),
        TG_OP,
        v_old_data,
        v_new_data,
        v_changed_fields,
        v_user_id
    );

    IF TG_OP = 'DELETE' THEN
        RETURN OLD;
    ELSE
        RETURN NEW;
    END IF;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 주요 테이블에 감사 트리거 적용
CREATE TRIGGER audit_users_changes
    AFTER INSERT OR UPDATE OR DELETE ON users
    FOR EACH ROW EXECUTE FUNCTION audit_log_changes();

CREATE TRIGGER audit_notes_changes
    AFTER INSERT OR UPDATE OR DELETE ON notes
    FOR EACH ROW EXECUTE FUNCTION audit_log_changes();

CREATE TRIGGER audit_user_books_changes
    AFTER INSERT OR UPDATE OR DELETE ON user_books
    FOR EACH ROW EXECUTE FUNCTION audit_log_changes();

CREATE TRIGGER audit_groups_changes
    AFTER INSERT OR UPDATE OR DELETE ON groups
    FOR EACH ROW EXECUTE FUNCTION audit_log_changes();
```

```typescript
// app/api/admin/audit/route.ts (관리자 전용)
import { NextRequest, NextResponse } from 'next/server';
import { createServerClient } from '@/lib/supabase/server';

export async function GET(request: NextRequest) {
  const supabase = createServerClient();

  // 관리자 권한 확인
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  }

  // 관리자 여부 확인 (별도 테이블 또는 메타데이터)
  const { data: profile } = await supabase
    .from('users')
    .select('is_admin')
    .eq('id', user.id)
    .single();

  if (!profile?.is_admin) {
    return NextResponse.json({ error: 'Forbidden' }, { status: 403 });
  }

  // 쿼리 파라미터
  const { searchParams } = new URL(request.url);
  const tableName = searchParams.get('table');
  const recordId = searchParams.get('recordId');
  const userId = searchParams.get('userId');
  const action = searchParams.get('action');
  const limit = parseInt(searchParams.get('limit') || '50');
  const offset = parseInt(searchParams.get('offset') || '0');

  // 감사 로그 조회
  let query = supabase
    .from('audit_logs')
    .select('*', { count: 'exact' })
    .order('created_at', { ascending: false })
    .range(offset, offset + limit - 1);

  if (tableName) query = query.eq('table_name', tableName);
  if (recordId) query = query.eq('record_id', recordId);
  if (userId) query = query.eq('user_id', userId);
  if (action) query = query.eq('action', action);

  const { data, error, count } = await query;

  if (error) throw error;

  return NextResponse.json({
    data,
    pagination: {
      total: count,
      limit,
      offset,
    },
  });
}
```
