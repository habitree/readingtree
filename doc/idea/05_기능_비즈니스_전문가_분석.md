## 3. 세부 기능 개발 방안

### 3.1 시맨틱 검색 (Vector Search)

#### 현재 문제
- 키워드 일치 검색만 가능
- 유사 내용 검색 불가
- "동기부여에 관한 기록" 같은 의미 검색 불가

#### 개선 방안: pgvector + OpenAI Embeddings

```sql
-- Supabase에 pgvector 확장 활성화
CREATE EXTENSION IF NOT EXISTS vector;

-- 노트에 임베딩 벡터 컬럼 추가
ALTER TABLE notes
ADD COLUMN embedding vector(1536);

-- 벡터 검색 인덱스
CREATE INDEX ON notes
USING ivfflat (embedding vector_cosine_ops)
WITH (lists = 100);

-- 시맨틱 검색 함수
CREATE OR REPLACE FUNCTION search_notes_semantic(
    query_embedding vector(1536),
    match_threshold float DEFAULT 0.7,
    match_count int DEFAULT 10
)
RETURNS TABLE (
    id uuid,
    content text,
    similarity float
)
LANGUAGE plpgsql
AS $$
BEGIN
    RETURN QUERY
    SELECT
        notes.id,
        notes.content,
        1 - (notes.embedding <=> query_embedding) as similarity
    FROM notes
    WHERE 1 - (notes.embedding <=> query_embedding) > match_threshold
    ORDER BY notes.embedding <=> query_embedding
    LIMIT match_count;
END;
$$;
```

#### 임베딩 생성 및 검색 API

```typescript
// lib/ai/embeddings.ts
import OpenAI from 'openai';

const openai = new OpenAI({
  apiKey: process.env.OPENAI_API_KEY,
});

export async function generateEmbedding(text: string): Promise<number[]> {
  const response = await openai.embeddings.create({
    model: 'text-embedding-3-small',
    input: text,
  });

  return response.data[0].embedding;
}

// 노트 저장 시 임베딩 생성 (Inngest 백그라운드 작업)
export const generateNoteEmbedding = inngest.createFunction(
  {
    id: 'generate-note-embedding',
    retries: 3,
  },
  { event: 'note/created' },
  async ({ event, step }) => {
    const { noteId, content } = event.data;

    // 임베딩 생성
    const embedding = await step.run('generate-embedding', async () => {
      return generateEmbedding(content);
    });

    // DB 업데이트
    await step.run('save-embedding', async () => {
      const supabase = createAdminClient();
      await supabase
        .from('notes')
        .update({ embedding })
        .eq('id', noteId);
    });
  }
);
```

#### 시맨틱 검색 API

```typescript
// app/api/search/semantic/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { generateEmbedding } from '@/lib/ai/embeddings';
import { createServerSupabaseClient } from '@/lib/supabase/server';

export async function GET(request: NextRequest) {
  const { searchParams } = new URL(request.url);
  const query = searchParams.get('q');
  const threshold = parseFloat(searchParams.get('threshold') || '0.7');
  const limit = parseInt(searchParams.get('limit') || '10');

  if (!query) {
    return NextResponse.json({ error: 'Query required' }, { status: 400 });
  }

  const supabase = createServerSupabaseClient();

  // 인증 확인
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
  }

  // 쿼리 임베딩 생성
  const queryEmbedding = await generateEmbedding(query);

  // 시맨틱 검색 실행
  const { data, error } = await supabase.rpc('search_notes_semantic', {
    query_embedding: queryEmbedding,
    match_threshold: threshold,
    match_count: limit,
  });

  if (error) throw error;

  return NextResponse.json({ results: data });
}
```

### 3.2 읽기 진행률 추적

#### 현재 문제
- 책 상태가 읽는 중/완독/중단 3가지뿐
- 진행 상황 파악 어려움
- 동기부여 요소 부족

#### 개선 방안: 페이지 기반 진행률

```sql
-- user_books 테이블 수정
ALTER TABLE user_books
ADD COLUMN total_pages INTEGER,
ADD COLUMN current_page INTEGER DEFAULT 0,
ADD COLUMN last_read_at TIMESTAMP WITH TIME ZONE;

-- 진행률 계산 뷰
CREATE VIEW user_books_with_progress AS
SELECT
    ub.*,
    CASE
        WHEN ub.total_pages > 0 THEN
            ROUND((ub.current_page::DECIMAL / ub.total_pages) * 100, 1)
        ELSE 0
    END as progress_percentage,
    b.title,
    b.author,
    b.cover_image_url
FROM user_books ub
JOIN books b ON ub.book_id = b.id;
```

#### 진행률 업데이트 컴포넌트

```typescript
// components/books/reading-progress-updater.tsx
'use client';

import { useState, useCallback } from 'react';
import { Slider } from '@/components/ui/slider';
import { Button } from '@/components/ui/button';
import { Input } from '@/components/ui/input';
import { BookOpen } from 'lucide-react';

interface ReadingProgressUpdaterProps {
  userBookId: string;
  currentPage: number;
  totalPages: number;
  onUpdate: (page: number) => Promise<void>;
}

export function ReadingProgressUpdater({
  userBookId,
  currentPage,
  totalPages,
  onUpdate,
}: ReadingProgressUpdaterProps) {
  const [page, setPage] = useState(currentPage);
  const [isUpdating, setIsUpdating] = useState(false);

  const progress = totalPages > 0 ? (page / totalPages) * 100 : 0;

  const handleUpdate = useCallback(async () => {
    setIsUpdating(true);
    try {
      await onUpdate(page);
    } finally {
      setIsUpdating(false);
    }
  }, [page, onUpdate]);

  return (
    <div className="p-4 bg-card rounded-lg border space-y-4">
      <div className="flex items-center justify-between">
        <h3 className="font-medium flex items-center gap-2">
          <BookOpen className="h-4 w-4" />
          읽기 진행률
        </h3>
        <span className="text-2xl font-bold text-primary">
          {progress.toFixed(0)}%
        </span>
      </div>

      {/* 프로그레스 바 */}
      <div className="h-3 bg-muted rounded-full overflow-hidden">
        <div
          className="h-full bg-primary transition-all duration-300"
          style={{ width: `${progress}%` }}
        />
      </div>

      {/* 페이지 입력 */}
      <div className="flex items-center gap-4">
        <div className="flex items-center gap-2 flex-1">
          <Input
            type="number"
            value={page}
            onChange={(e) => setPage(parseInt(e.target.value) || 0)}
            min={0}
            max={totalPages}
            className="w-24"
          />
          <span className="text-muted-foreground">/ {totalPages} 페이지</span>
        </div>

        <Button
          onClick={handleUpdate}
          disabled={isUpdating || page === currentPage}
          size="sm"
        >
          {isUpdating ? '저장 중...' : '업데이트'}
        </Button>
      </div>

      {/* 슬라이더 */}
      <Slider
        value={[page]}
        onValueChange={([value]) => setPage(value)}
        max={totalPages}
        step={1}
        className="mt-2"
      />

      {/* 완독까지 예상 */}
      {page > 0 && page < totalPages && (
        <p className="text-sm text-muted-foreground">
          남은 페이지: {totalPages - page}페이지
        </p>
      )}
    </div>
  );
}
```

### 3.3 독서 리포트 자동 생성

#### 현재 문제
- 기록은 있지만 정리/요약 없음
- 책 완독 후 인사이트 파악 어려움
- 기록의 가치 활용 부족

#### 개선 방안: AI 기반 독서 리포트

```typescript
// lib/ai/report-generator.ts
import { GoogleGenerativeAI } from '@google/generative-ai';
import { createAdminClient } from '@/lib/supabase/admin';

const genAI = new GoogleGenerativeAI(process.env.GEMINI_API_KEY!);

interface BookReport {
  summary: string;        // 책 요약
  keyInsights: string[];  // 핵심 인사이트
  favoriteQuotes: string[]; // 인상 깊은 구절
  themes: string[];       // 주요 테마
  personalReflection: string; // 개인적 성찰
  recommendedFor: string; // 추천 대상
}

export async function generateBookReport(
  bookId: string,
  userId: string
): Promise<BookReport> {
  const supabase = createAdminClient();

  // 책 정보 및 기록 조회
  const [bookResult, notesResult] = await Promise.all([
    supabase.from('books').select('*').eq('id', bookId).single(),
    supabase
      .from('notes')
      .select('*')
      .eq('book_id', bookId)
      .eq('user_id', userId)
      .order('page_number', { ascending: true }),
  ]);

  const book = bookResult.data;
  const notes = notesResult.data || [];

  // 기록 내용 정리
  const notesContent = notes
    .map((n) => {
      const pageInfo = n.page_number ? `(p.${n.page_number})` : '';
      return `[${n.type}${pageInfo}] ${n.content}`;
    })
    .join('\n\n');

  // Gemini로 리포트 생성
  const model = genAI.getGenerativeModel({ model: 'gemini-1.5-pro' });

  const prompt = `
    당신은 독서 분석 전문가입니다. 다음 책과 독서 기록을 분석하여 독서 리포트를 작성해주세요.

    책 정보:
    - 제목: ${book.title}
    - 저자: ${book.author}
    - 출판사: ${book.publisher}

    독서 기록:
    ${notesContent}

    다음 형식의 JSON으로 리포트를 작성해주세요:
    {
      "summary": "책의 핵심 내용 요약 (200자 내외)",
      "keyInsights": ["핵심 인사이트 1", "핵심 인사이트 2", "핵심 인사이트 3"],
      "favoriteQuotes": ["인상 깊은 구절 1", "인상 깊은 구절 2"],
      "themes": ["주요 테마 1", "주요 테마 2", "주요 테마 3"],
      "personalReflection": "독자의 기록을 바탕으로 한 개인적 성찰 제안 (100자 내외)",
      "recommendedFor": "이 책을 추천할 대상 (50자 내외)"
    }

    JSON만 출력하고 다른 설명은 포함하지 마세요.
  `;

  const result = await model.generateContent(prompt);
  const responseText = result.response.text();

  // JSON 파싱
  const jsonMatch = responseText.match(/\{[\s\S]*\}/);
  if (!jsonMatch) {
    throw new Error('Failed to parse report JSON');
  }

  return JSON.parse(jsonMatch[0]) as BookReport;
}
```

// lib/gamification/badge-checker.ts
import { createAdminClient } from '@/lib/supabase/admin';

interface BadgeCheck {
  badgeId: string;
  requirementType: string;
  requirementValue: number;

// 배지 획득 조건 확인
export async function checkAndAwardBadges(userId: string): Promise<string[]> {
  const supabase = createAdminClient();
  const earnedBadges: string[] = [];

  // 사용자 통계 조회
  const stats = await getUserStats(userId);

  // 모든 배지 정의 조회
  const { data: badges } = await supabase
    .from('badge_definitions')
    .select('*');

  // 이미 획득한 배지 조회
  const { data: userBadges } = await supabase
    .from('user_badges')
    .select('badge_id')
    .eq('user_id', userId);

  const earnedBadgeIds = new Set(userBadges?.map((b) => b.badge_id) || []);

  // 각 배지 조건 확인
  for (const badge of badges || []) {
    if (earnedBadgeIds.has(badge.id)) continue;

    const statValue = stats[badge.requirement_type] || 0;

    if (statValue >= badge.requirement_value) {
      // 배지 부여
      await supabase.from('user_badges').insert({
        user_id: userId,
        badge_id: badge.id,
      });

      // 포인트 추가
      await supabase.rpc('add_user_points', {
        p_user_id: userId,
        p_points: badge.points,
      });

      earnedBadges.push(badge.id);

      // 알림 발송
      await supabase.from('notifications').insert({
        user_id: userId,
        type: 'badge_earned',
        title: '새 배지 획득!',
        message: `"${badge.name}" 배지를 획득했습니다!`,
        data: { badgeId: badge.id },
      });
    }
  }

  return earnedBadges;
}

// lib/subscription/feature-flags.ts
import { createServerSupabaseClient } from '@/lib/supabase/server';

type SubscriptionTier = 'free' | 'pro' | 'team';

interface FeatureLimits {
  maxBooks: number;
  maxNotes: number;
  aiReportEnabled: boolean;
  semanticSearchEnabled: boolean;
  advancedStatsEnabled: boolean;
  adsEnabled: boolean;
  prioritySupport: boolean;
}

const FEATURE_LIMITS: Record<SubscriptionTier, FeatureLimits> = {
  free: {
    maxBooks: 50,
    maxNotes: 100,
    aiReportEnabled: false,
    semanticSearchEnabled: false,
    advancedStatsEnabled: false,
    adsEnabled: true,
    prioritySupport: false,
  },
  pro: {
    maxBooks: Infinity,
    maxNotes: Infinity,
    aiReportEnabled: true,
    semanticSearchEnabled: true,
    advancedStatsEnabled: true,
    adsEnabled: false,
    prioritySupport: true,
  },
  team: {
    maxBooks: Infinity,
    maxNotes: Infinity,
    aiReportEnabled: true,
    semanticSearchEnabled: true,
    advancedStatsEnabled: true,
    adsEnabled: false,
    prioritySupport: true,
  },
};

export async function getUserSubscription(userId: string): Promise<{
  tier: SubscriptionTier;
  limits: FeatureLimits;
  expiresAt?: Date;
}> {
  const supabase = createServerSupabaseClient();

  const { data: subscription } = await supabase
    .from('subscriptions')
    .select('*')
    .eq('user_id', userId)
    .eq('status', 'active')
    .single();

  const tier = (subscription?.tier as SubscriptionTier) || 'free';

  return {
    tier,
    limits: FEATURE_LIMITS[tier],
    expiresAt: subscription?.expires_at
      ? new Date(subscription.expires_at)
      : undefined,
  };
}

export async function checkFeatureAccess(
  userId: string,
  feature: keyof FeatureLimits
): Promise<boolean> {
  const { limits } = await getUserSubscription(userId);
  return !!limits[feature];
}

export async function checkUsageLimit(
  userId: string,
  type: 'books' | 'notes'
): Promise<{ allowed: boolean; current: number; limit: number }> {
  const supabase = createServerSupabaseClient();
  const { limits } = await getUserSubscription(userId);

  const limit = type === 'books' ? limits.maxBooks : limits.maxNotes;

  // 현재 사용량 조회
  const table = type === 'books' ? 'user_books' : 'notes';
  const { count } = await supabase
    .from(table)
    .select('*', { count: 'exact', head: true })
    .eq('user_id', userId);

  return {
    allowed: (count || 0) < limit,
    current: count || 0,
    limit,
  };
}

## 부록: KPI 대시보드 설계

### A. 핵심 지표

```sql
-- 일일 활성 사용자
SELECT COUNT(DISTINCT user_id)
FROM notes
WHERE created_at >= CURRENT_DATE;

-- 구독 전환율
SELECT
    COUNT(*) FILTER (WHERE tier = 'pro') * 100.0 / COUNT(*) as pro_conversion_rate
FROM users;

-- 평균 기록 수
SELECT AVG(note_count)
FROM (
    SELECT user_id, COUNT(*) as note_count
    FROM notes
    GROUP BY user_id
) t;
```

### B. 코호트 분석 쿼리

```sql
-- 주간 리텐션
WITH cohorts AS (
    SELECT
        user_id,
        DATE_TRUNC('week', MIN(created_at)) as cohort_week
    FROM notes
    GROUP BY user_id
),
activity AS (
    SELECT
        user_id,
        DATE_TRUNC('week', created_at) as activity_week
    FROM notes
)
SELECT
    c.cohort_week,
    COUNT(DISTINCT c.user_id) as cohort_size,
    COUNT(DISTINCT CASE WHEN a.activity_week = c.cohort_week + INTERVAL '1 week' THEN a.user_id END) as week_1,
    COUNT(DISTINCT CASE WHEN a.activity_week = c.cohort_week + INTERVAL '2 weeks' THEN a.user_id END) as week_2,
    COUNT(DISTINCT CASE WHEN a.activity_week = c.cohort_week + INTERVAL '4 weeks' THEN a.user_id END) as week_4
FROM cohorts c
LEFT JOIN activity a ON c.user_id = a.user_id
GROUP BY c.cohort_week
ORDER BY c.cohort_week DESC;
```
