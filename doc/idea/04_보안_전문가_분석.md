#### 개선 방안: @supabase/ssr 전환

```typescript
// lib/supabase/server.ts
import { createServerClient, type CookieOptions } from '@supabase/ssr';
import { cookies } from 'next/headers';
import { Database } from '@/types/database';

export function createServerSupabaseClient() {
  const cookieStore = cookies();

  return createServerClient<Database>(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    {
      cookies: {
        get(name: string) {
          return cookieStore.get(name)?.value;
        },
        set(name: string, value: string, options: CookieOptions) {
          try {
            cookieStore.set({ name, value, ...options });
          } catch (error) {
            // Server Component에서는 쿠키 설정 불가
          }
        },
        remove(name: string, options: CookieOptions) {
          try {
            cookieStore.set({ name, value: '', ...options });
          } catch (error) {
            // Server Component에서는 쿠키 삭제 불가
          }
        },
      },
    }
  );
}
```

```typescript
import { NextResponse, type NextRequest } from 'next/server';

            },
          });

            value,
            ...options,
          });
        },

            name,
            value: '',

            request: {
              headers: request.headers,
            },
          });
          response.cookies.set({
            name,
            value: '',
            ...options,
          });
        },
      },
    }
  );

  // 세션 갱신
  const { data: { user } } = await supabase.auth.getUser();

  return { response, user };
```

#### 미들웨어 통합

```typescript
// middleware.ts
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';
import { updateSession } from '@/lib/supabase/middleware';
import {
  ipRateLimiter,
  authRateLimiter,
  userRateLimiter,
} from '@/lib/security/rate-limit';

// Rate Limit 헤더 추가 함수
function addRateLimitHeaders(
  response: NextResponse,
  result: { limit: number; remaining: number; reset: number }
) {
  response.headers.set('X-RateLimit-Limit', result.limit.toString());
  response.headers.set('X-RateLimit-Remaining', result.remaining.toString());
  response.headers.set('X-RateLimit-Reset', result.reset.toString());
  return response;
}

export async function middleware(request: NextRequest) {
  const ip = request.ip ?? request.headers.get('x-forwarded-for') ?? 'unknown';
  const path = request.nextUrl.pathname;

  // 1. IP 기반 글로벌 Rate Limit
  const ipResult = await ipRateLimiter.limit(ip);
  if (!ipResult.success) {
    return addRateLimitHeaders(
      NextResponse.json(
        { error: 'Too many requests. Please try again later.' },
        { status: 429 }
      ),
      ipResult
    );
  }

  // 2. 인증 엔드포인트 강화 Rate Limit
  if (path.startsWith('/api/auth') || path.startsWith('/login')) {
    const authResult = await authRateLimiter.limit(ip);
    if (!authResult.success) {
      // 의심스러운 활동 로깅
      console.warn(`Auth rate limit exceeded: IP=${ip}, path=${path}`);

      return addRateLimitHeaders(
        NextResponse.json(
          {
            error: 'Too many authentication attempts. Please wait 15 minutes.',
            retryAfter: Math.ceil((authResult.reset - Date.now()) / 1000),
          },
          { status: 429 }
        ),
        authResult
      );
    }
  }

  // 3. 세션 업데이트 및 사용자 확인
  const { response, user } = await updateSession(request);

  // 4. 인증된 사용자 Rate Limit
  if (user && path.startsWith('/api/')) {
    const userResult = await userRateLimiter.limit(user.id);
    if (!userResult.success) {
      return addRateLimitHeaders(
        NextResponse.json(
          { error: 'Rate limit exceeded. Please slow down.' },
          { status: 429 }
        ),
        userResult
      );
    }
  }

  // 5. 보호된 경로 접근 제어
  const protectedPaths = ['/books', '/notes', '/groups', '/profile', '/timeline'];
  const isProtectedPath = protectedPaths.some((p) => path.startsWith(p));

  if (isProtectedPath && !user) {
    const redirectUrl = new URL('/login', request.url);
    redirectUrl.searchParams.set('redirectTo', path);
    return NextResponse.redirect(redirectUrl);
  }

  return addRateLimitHeaders(response, ipResult);
}

export const config = {
  matcher: [
    '/((?!_next/static|_next/image|favicon.ico|.*\\.(?:svg|png|jpg|jpeg|gif|webp)$).*)',
  ],
};
```

### 3.4 입력 검증 강화 (파일 업로드)

#### 현재 문제
- 파일 타입 검증 불충분
- 파일 크기 서버 측 검증 미흡
- 악성 파일 업로드 가능성

#### 개선 방안: 다층 파일 검증

```typescript
// lib/security/file-validation.ts
import { z } from 'zod';

// 허용된 MIME 타입
const ALLOWED_IMAGE_TYPES = [
  'image/jpeg',
  'image/png',
  'image/webp',
  'image/gif',
] as const;

// 파일 시그니처 (Magic Bytes)
const FILE_SIGNATURES: Record<string, number[]> = {
  'image/jpeg': [0xFF, 0xD8, 0xFF],
  'image/png': [0x89, 0x50, 0x4E, 0x47],
  'image/gif': [0x47, 0x49, 0x46],
  'image/webp': [0x52, 0x49, 0x46, 0x46], // RIFF
};

// 최대 파일 크기 (5MB)
const MAX_FILE_SIZE = 5 * 1024 * 1024;

// 파일 업로드 스키마
export const fileUploadSchema = z.object({
  file: z.instanceof(File).refine(
    (file) => file.size <= MAX_FILE_SIZE,
    `파일 크기는 최대 ${MAX_FILE_SIZE / 1024 / 1024}MB입니다`
  ),
  type: z.enum(['photo', 'transcription', 'avatar']),
});

// 파일 시그니처 검증
export async function validateFileSignature(file: File): Promise<boolean> {
  const buffer = await file.slice(0, 12).arrayBuffer();
  const bytes = new Uint8Array(buffer);

  const expectedSignature = FILE_SIGNATURES[file.type];
  if (!expectedSignature) {
    return false;
  }

  // WebP 특수 처리 (RIFF....WEBP)
  if (file.type === 'image/webp') {
    const riff = bytes.slice(0, 4);
    const webp = bytes.slice(8, 12);
    return (
      riff[0] === 0x52 && riff[1] === 0x49 && riff[2] === 0x46 && riff[3] === 0x46 &&
      webp[0] === 0x57 && webp[1] === 0x45 && webp[2] === 0x42 && webp[3] === 0x50
    );
  }

  return expectedSignature.every((byte, index) => bytes[index] === byte);
}

// 파일명 정규화 (Path Traversal 방지)
export function sanitizeFileName(fileName: string): string {
  // 경로 구분자 제거
  let sanitized = fileName.replace(/[\/\\]/g, '');

  // 특수 문자 제거
  sanitized = sanitized.replace(/[^a-zA-Z0-9가-힣._-]/g, '_');

  // 숨김 파일 방지
  if (sanitized.startsWith('.')) {
    sanitized = '_' + sanitized;
  }

  // 길이 제한
  if (sanitized.length > 100) {
    const ext = sanitized.split('.').pop() || '';
    sanitized = sanitized.slice(0, 95) + '.' + ext;
  }

  return sanitized;
}

// 통합 파일 검증
export async function validateUploadedFile(
  file: File,
  allowedTypes: readonly string[] = ALLOWED_IMAGE_TYPES
): Promise<{ valid: boolean; error?: string }> {
  // 1. MIME 타입 검사
  if (!allowedTypes.includes(file.type)) {
    return {
      valid: false,
      error: `허용되지 않는 파일 형식입니다. 허용: ${allowedTypes.join(', ')}`,
    };
  }

  // 2. 파일 크기 검사
  if (file.size > MAX_FILE_SIZE) {
    return {
      valid: false,
      error: `파일 크기가 너무 큽니다. 최대: ${MAX_FILE_SIZE / 1024 / 1024}MB`,
    };
  }

  // 3. 파일 시그니처 검증 (MIME spoofing 방지)
  const validSignature = await validateFileSignature(file);
  if (!validSignature) {
    return {
      valid: false,
      error: '파일 형식이 올바르지 않습니다. 실제 이미지 파일을 업로드해주세요.',
    };
  }

  // 4. 파일명 검증
  const sanitizedName = sanitizeFileName(file.name);
  if (sanitizedName !== file.name) {
    console.warn(`파일명 정규화: ${file.name} -> ${sanitizedName}`);
  }

  return { valid: true };
}
```

#### 파일 업로드 API에 적용

```typescript
// app/api/upload/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { createServerSupabaseClient } from '@/lib/supabase/server';
import {
  validateUploadedFile,
  sanitizeFileName,
  fileUploadSchema,
} from '@/lib/security/file-validation';
import sharp from 'sharp';

export async function POST(request: NextRequest) {
  try {
    const supabase = createServerSupabaseClient();

    // 인증 확인
    const { data: { user }, error: authError } = await supabase.auth.getUser();
    if (authError || !user) {
      return NextResponse.json(
        { error: 'Unauthorized' },
        { status: 401 }
      );
    }

    // 폼 데이터 파싱
    const formData = await request.formData();
    const file = formData.get('file') as File | null;
    const type = formData.get('type') as string | null;

    if (!file || !type) {
      return NextResponse.json(
        { error: 'File and type are required' },
        { status: 400 }
      );
    }

    // 스키마 검증
    const schemaResult = fileUploadSchema.safeParse({ file, type });
    if (!schemaResult.success) {
      return NextResponse.json(
        { error: schemaResult.error.errors[0].message },
        { status: 400 }
      );
    }

    // 파일 검증 (MIME + 시그니처 + 크기)
    const validationResult = await validateUploadedFile(file);
    if (!validationResult.valid) {
      return NextResponse.json(
        { error: validationResult.error },
        { status: 400 }
      );
    }

    // 파일명 정규화
    const sanitizedFileName = sanitizeFileName(file.name);

    // 이미지 처리 (메타데이터 제거 + 크기 최적화)
    const buffer = await file.arrayBuffer();
    const processedImage = await sharp(Buffer.from(buffer))
      .rotate() // EXIF 회전 정보 적용
      .resize(1920, 1920, { fit: 'inside', withoutEnlargement: true })
      .jpeg({ quality: 85 })
      .toBuffer();

    // 고유 파일명 생성
    const timestamp = Date.now();
    const random = Math.random().toString(36).substring(7);
    const ext = 'jpg'; // 항상 JPEG로 변환
    const fileName = `${timestamp}_${random}.${ext}`;
    const filePath = `${type}s/${user.id}/${fileName}`;

    // Supabase Storage 업로드
    const { data, error } = await supabase.storage
      .from('images')
      .upload(filePath, processedImage, {
        contentType: 'image/jpeg',
        cacheControl: '3600',
        upsert: false,
      });

    if (error) {
      console.error('Upload error:', error);
      return NextResponse.json(
        { error: 'Failed to upload file' },
        { status: 500 }
      );
    }

    // 공개 URL 생성
    const { data: { publicUrl } } = supabase.storage
      .from('images')
      .getPublicUrl(filePath);

    return NextResponse.json({
      url: publicUrl,
      path: filePath,
    });

  } catch (error) {
    console.error('Upload handler error:', error);
    return NextResponse.json(
      { error: 'Internal server error' },
      { status: 500 }
    );
  }
}
```

#### 현재 문제

#### 개선 방안: 계정 데이터 관리 기능

```typescript
// lib/api/user-data.ts
import { createAdminClient } from '@/lib/supabase/admin';

interface UserDataExport {
  user: any;
  books: any[];
  notes: any[];
  groups: any[];
}

// 사용자 데이터 내보내기
export async function exportUserData(userId: string): Promise<UserDataExport> {
  const supabase = createAdminClient();

  // 병렬로 모든 데이터 조회
  const [userResult, booksResult, notesResult, groupsResult] = await Promise.all([
    supabase.from('users').select('*').eq('id', userId).single(),
    supabase.from('user_books').select('*, book:books(*)').eq('user_id', userId),
    supabase.from('notes').select('*').eq('user_id', userId),
    supabase.from('group_members').select('*, group:groups(*)').eq('user_id', userId),
  ]);

  return {
    user: userResult.data,
    books: booksResult.data || [],
    notes: notesResult.data || [],
    groups: groupsResult.data || [],
  };
}

// 사용자 데이터 완전 삭제
export async function deleteUserData(userId: string): Promise<void> {
  const supabase = createAdminClient();

  // 1. Storage 파일 삭제
  const storageFolders = ['photos', 'transcriptions', 'avatars'];

  for (const folder of storageFolders) {
    const { data: files } = await supabase.storage
      .from('images')
      .list(`${folder}/${userId}`);

    if (files && files.length > 0) {
      const filePaths = files.map((f) => `${folder}/${userId}/${f.name}`);
      await supabase.storage.from('images').remove(filePaths);
    }
  }

  // 2. 관련 데이터 삭제 (CASCADE로 자동 삭제되지 않는 항목)
  // 그룹 노트 공유 해제
  await supabase
    .from('group_notes')
    .delete()
    .in(
      'note_id',
      supabase.from('notes').select('id').eq('user_id', userId)
    );

  // 3. 그룹 리더인 경우 처리
  const { data: leadingGroups } = await supabase
    .from('groups')
    .select('id')
    .eq('leader_id', userId);

  if (leadingGroups && leadingGroups.length > 0) {
    for (const group of leadingGroups) {
      // 다른 멤버에게 리더 위임 또는 그룹 삭제
      const { data: otherMembers } = await supabase
        .from('group_members')
        .select('user_id')
        .eq('group_id', group.id)
        .neq('user_id', userId)
        .eq('status', 'approved')
        .limit(1);

      if (otherMembers && otherMembers.length > 0) {
        // 리더 위임
        await supabase
          .from('groups')
          .update({ leader_id: otherMembers[0].user_id })
          .eq('id', group.id);

        await supabase
          .from('group_members')
          .update({ role: 'leader' })
          .eq('group_id', group.id)
          .eq('user_id', otherMembers[0].user_id);
      } else {
        // 멤버 없으면 그룹 삭제
        await supabase.from('groups').delete().eq('id', group.id);
      }
    }
  }

  // 4. 사용자 프로필 삭제 (CASCADE로 notes, user_books 등 자동 삭제)
  await supabase.from('users').delete().eq('id', userId);

  // 5. Supabase Auth 사용자 삭제
  await supabase.auth.admin.deleteUser(userId);

  // 6. 감사 로그 기록 (익명화)
  await supabase.from('audit_logs').insert({
    table_name: 'users',
    record_id: userId,
    action: 'GDPR_DELETE',
    old_data: { anonymized: true },
    new_data: null,
    user_id: null, // 삭제된 사용자이므로 null
  });
}
```

```typescript
// app/api/user/delete-account/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { createServerSupabaseClient } from '@/lib/supabase/server';
import { exportUserData, deleteUserData } from '@/lib/api/user-data';
import { z } from 'zod';

const deleteAccountSchema = z.object({
  confirmation: z.literal('DELETE MY ACCOUNT'),
  reason: z.string().optional(),
});

export async function POST(request: NextRequest) {
  try {
    const supabase = createServerSupabaseClient();

    // 인증 확인
    const { data: { user }, error: authError } = await supabase.auth.getUser();
    if (authError || !user) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }

    // 요청 검증
    const body = await request.json();
    const result = deleteAccountSchema.safeParse(body);

    if (!result.success) {
      return NextResponse.json(
        { error: '계정 삭제를 확인하려면 "DELETE MY ACCOUNT"를 입력해주세요.' },
        { status: 400 }
      );
    }

    // 데이터 내보내기 (선택사항 - 사용자에게 제공)
    const exportedData = await exportUserData(user.id);

    // 삭제 사유 로깅 (통계용)
    if (result.data.reason) {
      console.log(`Account deletion reason [${user.id}]: ${result.data.reason}`);
    }

    // 데이터 삭제 실행
    await deleteUserData(user.id);

    // 성공 응답
    return NextResponse.json({
      success: true,
      message: '계정이 삭제되었습니다. 그동안 ReadTree를 이용해주셔서 감사합니다.',
      exportedData, // 삭제 전 데이터 반환
    });

  } catch (error) {
    console.error('Account deletion error:', error);
    return NextResponse.json(
      { error: '계정 삭제 중 오류가 발생했습니다. 고객센터에 문의해주세요.' },
      { status: 500 }
    );
  }
}
```

```typescript
// lib/security/audit-logger.ts
import { createAdminClient } from '@/lib/supabase/admin';

type SecurityEventType =
  | 'AUTH_LOGIN_SUCCESS'
  | 'AUTH_LOGIN_FAILED'
  | 'AUTH_LOGOUT'
  | 'AUTH_PASSWORD_RESET'
  | 'AUTH_MFA_ENABLED'
  | 'AUTH_MFA_DISABLED'
  | 'RATE_LIMIT_EXCEEDED'
  | 'SUSPICIOUS_ACTIVITY'
  | 'DATA_EXPORT'
  | 'ACCOUNT_DELETED'
  | 'PERMISSION_DENIED'
  | 'FILE_UPLOAD_BLOCKED'
  | 'CSP_VIOLATION';

interface SecurityEvent {
  type: SecurityEventType;
  userId?: string;
  ipAddress: string;
  userAgent: string;
  details: Record<string, any>;
  severity: 'low' | 'medium' | 'high' | 'critical';
}

export async function logSecurityEvent(event: SecurityEvent): Promise<void> {
  const supabase = createAdminClient();

  const logEntry = {
    event_type: event.type,
    user_id: event.userId || null,
    ip_address: event.ipAddress,
    user_agent: event.userAgent,
    details: event.details,
    severity: event.severity,
    created_at: new Date().toISOString(),
  };

  // DB에 저장
  await supabase.from('security_logs').insert(logEntry);

  // 심각한 이벤트는 알림 발송
  if (event.severity === 'high' || event.severity === 'critical') {
    await sendSecurityAlert(event);
  }
}

// 보안 알림 발송 (Slack, Email 등)
async function sendSecurityAlert(event: SecurityEvent): Promise<void> {
  // Slack Webhook (예시)
  const webhookUrl = process.env.SLACK_SECURITY_WEBHOOK;

  if (webhookUrl) {
    await fetch(webhookUrl, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        text: `🚨 Security Alert: ${event.type}`,
        attachments: [
          {
            color: event.severity === 'critical' ? '#ff0000' : '#ff9900',
            fields: [
              { title: 'Type', value: event.type, short: true },
              { title: 'Severity', value: event.severity, short: true },
              { title: 'IP', value: event.ipAddress, short: true },
              { title: 'User ID', value: event.userId || 'N/A', short: true },
              { title: 'Details', value: JSON.stringify(event.details) },
            ],
          },
        ],
      }),
    });
  }
}

// 의심스러운 활동 감지
export function detectSuspiciousActivity(
  events: Array<{ type: string; timestamp: Date; ip: string }>
): boolean {
  // 짧은 시간 내 여러 실패 시도
  const recentFailures = events.filter(
    (e) =>
      e.type === 'AUTH_LOGIN_FAILED' &&
      Date.now() - e.timestamp.getTime() < 5 * 60 * 1000 // 5분
  );

  if (recentFailures.length >= 5) {
    return true;
  }

  // 다른 IP에서의 동시 로그인 시도
  const uniqueIps = new Set(events.map((e) => e.ip));
  if (uniqueIps.size >= 3 && events.length > 10) {
    return true;
  }

  return false;
}
```
