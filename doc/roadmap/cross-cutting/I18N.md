## 1. 현황 분석

### 1.1 현재 상태

- 다국어 지원 없음
- 모든 텍스트 하드코딩

| 언어 | 코드 | 우선순위 | 시장 |

| 한국어 | ko | 기본 | 국내 |
| 영어 | en | 1순위 | 글로벌 |
| 일본어 | ja | 2순위 | 일본 |

## 2. 기술 스택

### 2.1 라이브러리 선택

```json
{
  "dependencies": {
    "next-intl": "^3.0.0"
  }
}
```

**next-intl 선택 이유:**
- Next.js App Router 네이티브 지원
- 서버 컴포넌트 지원
- 타입 안전성
- 작은 번들 사이즈

### 2.2 디렉토리 구조

```
messages/
├── ko.json               # 한국어 (기본)
├── en.json               # 영어
└── ja.json               # 일본어

app/
├── [locale]/             # 로케일 라우트
│   ├── layout.tsx
│   ├── page.tsx
│   └── (main)/
│       ├── books/
│       ├── notes/
│       └── ...

i18n.ts                   # i18n 설정
navigation.ts             # 로케일 네비게이션
```

## 3. 구현 설계

### 3.1 설정 파일

```typescript
// i18n.ts
import { getRequestConfig } from 'next-intl/server';
import { notFound } from 'next/navigation';

export const locales = ['ko', 'en', 'ja'] as const;
export const defaultLocale = 'ko';

export type Locale = (typeof locales)[number];

export default getRequestConfig(async ({ locale }) => {
  if (!locales.includes(locale as Locale)) notFound();

  return {
    messages: (await import(`./messages/${locale}.json`)).default,
    timeZone: 'Asia/Seoul',
    now: new Date(),
  };
});

// middleware.ts
import createMiddleware from 'next-intl/middleware';
import { locales, defaultLocale } from './i18n';

export default createMiddleware({
  locales,
  defaultLocale,
  localePrefix: 'as-needed', // 기본 언어는 URL에 표시 안 함
});

export const config = {
  matcher: ['/((?!api|_next|_vercel|.*\\..*).*)'],
};
```

### 3.3 컴포넌트에서 사용

```typescript
// Server Component
import { getTranslations } from 'next-intl/server';

export default async function BooksPage() {
  const t = await getTranslations('books');

  return (
    <div>
      <h1>{t('title')}</h1>
      <Button>{t('addBook')}</Button>
    </div>
  );

// Client Component
'use client';

import { useTranslations } from 'next-intl';

export function BookCard({ book }: { book: Book }) {
  const t = useTranslations('books');

  return (
    <Card>
      <CardTitle>{book.title}</CardTitle>
      <Badge>{t(`status.${book.status}`)}</Badge>
      <p>{t('progress', { current: book.currentPage, total: book.totalPages })}</p>
    </Card>
  );
}
```

### 3.4 날짜/숫자 포맷

```typescript
// 날짜 포맷
import { useFormatter } from 'next-intl';

function DateDisplay({ date }: { date: Date }) {
  const format = useFormatter();

  return (
    <time dateTime={date.toISOString()}>
      {format.dateTime(date, {
        year: 'numeric',
        month: 'long',
        day: 'numeric',
      })}
    </time>
  );
}

// 상대 시간
function RelativeTime({ date }: { date: Date }) {
  const format = useFormatter();

  return (
    <time dateTime={date.toISOString()}>
      {format.relativeTime(date)}
    </time>
  );
}

// 숫자 포맷
function PointsDisplay({ points }: { points: number }) {
  const format = useFormatter();

  return (
    <span>
      {format.number(points, { style: 'decimal' })}P
    </span>
  );
}
```

### 3.5 언어 선택기

```typescript
'use client';

import { useLocale } from 'next-intl';
import { useRouter, usePathname } from 'next/navigation';

const languages = [
  { code: 'ko', name: '한국어', flag: '🇰🇷' },
  { code: 'en', name: 'English', flag: '🇺🇸' },
  { code: 'ja', name: '日本語', flag: '🇯🇵' },
];

export function LanguageSelector() {
  const locale = useLocale();
  const router = useRouter();
  const pathname = usePathname();

  const handleChange = (newLocale: string) => {
    // 현재 경로에서 로케일 변경
    const newPath = pathname.replace(`/${locale}`, `/${newLocale}`);
    router.push(newPath);
  };

  return (
    <DropdownMenu>
      <DropdownMenuTrigger asChild>
        <Button variant="ghost" size="sm">
          {languages.find(l => l.code === locale)?.flag}
          <ChevronDown className="ml-1 h-4 w-4" />
        </Button>
      </DropdownMenuTrigger>
      <DropdownMenuContent>
        {languages.map((lang) => (
          <DropdownMenuItem
            key={lang.code}
            onClick={() => handleChange(lang.code)}
          >
            <span className="mr-2">{lang.flag}</span>
            {lang.name}
            {locale === lang.code && <Check className="ml-auto h-4 w-4" />}
          </DropdownMenuItem>
        ))}
      </DropdownMenuContent>
    </DropdownMenu>
  );
}
```

## 4. 번역 워크플로우

### 4.1 번역 관리

```
번역 워크플로우
═══════════════════════════════════════════════════════════

  [개발자]                [번역가]              [검수자]
     │                       │                     │
     ▼                       │                     │
  ko.json 작성               │                     │
     │                       │                     │
     ▼                       │                     │
  번역 요청 ─────────────────▶                     │
     │                       ▼                     │
     │                  en.json 작성               │
     │                  ja.json 작성               │
     │                       │                     │
     │                       ▼                     │
     │               번역 완료 ─────────────────────▶
     │                       │                 검수
     │                       │                     │
     │◀──────────────────────┼─────────────────────
     │                                         피드백
     ▼
  번역 적용

═══════════════════════════════════════════════════════════
```

### 4.2 번역 키 추출 자동화

```typescript
// scripts/extract-i18n.ts
import fs from 'fs';
import path from 'path';
import glob from 'glob';

// 코드에서 번역 키 추출
const extractKeys = () => {
  const files = glob.sync('**/*.{ts,tsx}', {
    ignore: ['node_modules/**', '.next/**'],
  });

  const keys = new Set<string>();

  files.forEach(file => {
    const content = fs.readFileSync(file, 'utf-8');

    // t('key') 패턴 매칭
    const matches = content.matchAll(/t\(['"]([^'"]+)['"]\)/g);
    for (const match of matches) {
      keys.add(match[1]);
    }
  });

  return Array.from(keys).sort();
};

// 누락된 번역 키 확인
const findMissingKeys = () => {
  const extractedKeys = extractKeys();
  const messages = JSON.parse(
    fs.readFileSync('messages/ko.json', 'utf-8')
  );

  const existingKeys = flattenKeys(messages);
  const missing = extractedKeys.filter(key => !existingKeys.includes(key));

  if (missing.length > 0) {
    console.warn('Missing translation keys:', missing);
  }
};
```

```typescript
// app/[locale]/layout.tsx
import { getTranslations } from 'next-intl/server';

export async function generateMetadata({ params: { locale } }) {
  const t = await getTranslations({ locale, namespace: 'metadata' });

  return {
    title: {
      default: t('title'),
      template: `%s | ${t('title')}`,
    },
    description: t('description'),
    alternates: {
      languages: {
        ko: '/ko',
        en: '/en',
        ja: '/ja',
      },
    },
  };
}
```
