## 1. 현황 분석

### 1.1 Core Web Vitals 현황 (추정)

| 지표 | 권장 기준 | 현재 상태 (추정) | 목표 |
|------|-----------|-----------------|------|
| **LCP** (Largest Contentful Paint) | < 2.5s | ~3.5s | < 2.0s |
| **FID** (First Input Delay) | < 100ms | ~150ms | < 50ms |
| **CLS** (Cumulative Layout Shift) | < 0.1 | ~0.15 | < 0.05 |
| **TTFB** (Time to First Byte) | < 800ms | ~1.2s | < 500ms |
| **FCP** (First Contentful Paint) | < 1.8s | ~2.5s | < 1.5s |

### 1.2 현재 번들 구성 분석

```
현재 번들 구성 (추정)
├── Framework (Next.js, React)     ~150KB gzipped
├── UI Components (shadcn/ui)      ~80KB gzipped
├── Charts (Recharts)              ~120KB gzipped ⚠️ 큰 번들
├── Carousel (Swiper)              ~50KB gzipped
├── Form (React Hook Form, Zod)    ~30KB gzipped
├── Supabase Client                ~40KB gzipped
└── Application Code               ~50KB gzipped
────────────────────────────────────
Total (추정)                       ~520KB gzipped
```

### 1.3 현재 강점

- **Next.js App Router**: 자동 코드 스플리팅
- **TypeScript**: 컴파일 타임 최적화
- **Tailwind CSS**: 사용하지 않는 CSS 제거 (PurgeCSS)

### 1.4 개선 필요 영역

| 영역 | 이슈 | 영향 |
|------|------|------|
| 번들 크기 | Recharts 전체 로드 | LCP, FCP 저하 |
| 이미지 | WebP/AVIF 미적용 | LCP 저하 |
| 렌더링 | 불필요한 리렌더링 | FID 저하 |
| 데이터 페칭 | Waterfall 패턴 | TTFB 저하 |

```typescript
import sharp from 'sharp';

// 서버에서 blur placeholder 생성
export async function generateBlurPlaceholder(
  imageUrl: string
): Promise<string> {
  try {
    const response = await fetch(imageUrl);
    const buffer = await response.arrayBuffer();

    const { data, info } = await sharp(Buffer.from(buffer))
      .resize(10, 15, { fit: 'inside' })
      .blur()
      .toBuffer({ resolveWithObject: true });

    const base64 = `data:image/${info.format};base64,${data.toString('base64')}`;
    return base64;
  } catch {
    // 기본 placeholder 반환
    return 'data:image/jpeg;base64,/9j/4AAQSkZJRgABAQAAAQABAAD...';
  }
}
```

### 3.3 Header useEffect 통합

#### 현재 문제
- Header 컴포넌트에 여러 개의 useEffect
- 불필요한 리렌더링 발생
- 성능 저하 원인

#### 개선 전 코드 (문제점)

```typescript
// ❌ 개선 전: 여러 useEffect로 분산
function Header() {
  const [user, setUser] = useState(null);
  const [notifications, setNotifications] = useState([]);
  const [isScrolled, setIsScrolled] = useState(false);

  // useEffect #1: 사용자 정보
  useEffect(() => {
    fetchUser();
  }, []);

  // useEffect #2: 알림
  useEffect(() => {
    fetchNotifications();
  }, [user]);

  // useEffect #3: 스크롤 감지
  useEffect(() => {
    const handleScroll = () => {
      setIsScrolled(window.scrollY > 0);
    };
    window.addEventListener('scroll', handleScroll);
    return () => window.removeEventListener('scroll', handleScroll);
  }, []);

  // ... 3번의 리렌더링 유발 가능
```

#### 개선 후 코드

```typescript
// components/layout/header.tsx
'use client';

import { useCallback, useEffect, useReducer, useRef } from 'react';
import { createClientComponentClient } from '@supabase/auth-helpers-nextjs';

// 상태 통합
interface HeaderState {
  isScrolled: boolean;
  notificationCount: number;
  isMenuOpen: boolean;
}

type HeaderAction =
  | { type: 'SET_SCROLLED'; payload: boolean }
  | { type: 'SET_NOTIFICATION_COUNT'; payload: number }
  | { type: 'TOGGLE_MENU' };

function headerReducer(state: HeaderState, action: HeaderAction): HeaderState {
  switch (action.type) {
    case 'SET_SCROLLED':
      return { ...state, isScrolled: action.payload };
    case 'SET_NOTIFICATION_COUNT':
      return { ...state, notificationCount: action.payload };
    case 'TOGGLE_MENU':
      return { ...state, isMenuOpen: !state.isMenuOpen };
    default:
      return state;
  }
}

export function Header() {
  const [state, dispatch] = useReducer(headerReducer, {
    isScrolled: false,
    notificationCount: 0,
    isMenuOpen: false,
  });

  const supabase = createClientComponentClient();
  const scrollRAF = useRef<number>();

  // ✅ 통합된 useEffect
  useEffect(() => {
    // 1. 스크롤 핸들러 (throttled with RAF)
    const handleScroll = () => {
      if (scrollRAF.current) return;

      scrollRAF.current = requestAnimationFrame(() => {
        const isScrolled = window.scrollY > 10;
        dispatch({ type: 'SET_SCROLLED', payload: isScrolled });
        scrollRAF.current = undefined;
      });
    };

    // 2. 알림 구독
    const notificationChannel = supabase
      .channel('notifications')
      .on(
        'postgres_changes',
        {
          event: 'INSERT',
          schema: 'public',
          table: 'notifications',
        },
        (payload) => {
          dispatch({
            type: 'SET_NOTIFICATION_COUNT',
            payload: state.notificationCount + 1,
          });
        }
      )
      .subscribe();

    // 이벤트 리스너 등록
    window.addEventListener('scroll', handleScroll, { passive: true });

    // 클린업
    return () => {
      window.removeEventListener('scroll', handleScroll);
      if (scrollRAF.current) {
        cancelAnimationFrame(scrollRAF.current);
      }
      supabase.removeChannel(notificationChannel);
    };
  }, []); // 의존성 최소화

  const toggleMenu = useCallback(() => {
    dispatch({ type: 'TOGGLE_MENU' });
  }, []);

  return (
    <header
      className={cn(
        'sticky top-0 z-50 w-full border-b transition-all duration-200',
        state.isScrolled
          ? 'bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/60'
          : 'bg-background'
      )}
    >
      {/* ... */}
    </header>
  );
}
```

### 3.5 커서 기반 페이지네이션

#### 현재 문제
- Offset 기반 페이지네이션 사용
- 대용량 데이터에서 성능 저하
- 페이지 이동 시 일관성 문제

#### 개선 방안: 커서 기반 페이지네이션

```typescript
// lib/api/notes.ts
import { createServerClient } from '@/lib/supabase/server';

interface CursorPaginationParams {
  userId: string;
  cursor?: string; // 마지막 항목의 ID 또는 timestamp
  limit?: number;
  direction?: 'forward' | 'backward';
}

interface PaginatedResult<T> {
  data: T[];
  nextCursor: string | null;
  prevCursor: string | null;
  hasMore: boolean;
}

export async function getNotesByCursor({
  userId,
  cursor,
  limit = 20,
  direction = 'forward',
}: CursorPaginationParams): Promise<PaginatedResult<Note>> {
  const supabase = createServerClient();

  let query = supabase
    .from('notes')
    .select(`
      *,
      book:books(id, title, cover_image_url)
    `)
    .eq('user_id', userId)
    .order('created_at', { ascending: false })
    .limit(limit + 1); // 다음 페이지 존재 여부 확인용 +1

  // 커서가 있으면 해당 위치부터 조회
  if (cursor) {
    if (direction === 'forward') {
      query = query.lt('created_at', cursor);
    } else {
      query = query.gt('created_at', cursor);
    }
  }

  const { data, error } = await query;

  if (error) throw error;

  const hasMore = data.length > limit;
  const items = hasMore ? data.slice(0, -1) : data;

  return {
    data: items,
    nextCursor: hasMore ? items[items.length - 1].created_at : null,
    prevCursor: cursor || null,
    hasMore,
  };
}
```

#### 무한 스크롤 훅

```typescript
// hooks/use-infinite-notes.ts
'use client';

import { useState, useCallback, useRef, useEffect } from 'react';
import { Note } from '@/types';

interface UseInfiniteNotesOptions {
  initialCursor?: string;
  limit?: number;
}

export function useInfiniteNotes(options: UseInfiniteNotesOptions = {}) {
  const { limit = 20 } = options;

  const [notes, setNotes] = useState<Note[]>([]);
  const [cursor, setCursor] = useState<string | null>(null);
  const [hasMore, setHasMore] = useState(true);
  const [isLoading, setIsLoading] = useState(false);
  const [error, setError] = useState<Error | null>(null);

  const loadingRef = useRef(false);

  const loadMore = useCallback(async () => {
    if (loadingRef.current || !hasMore) return;

    loadingRef.current = true;
    setIsLoading(true);

    try {
      const response = await fetch(
        `/api/notes?cursor=${cursor || ''}&limit=${limit}`
      );

      if (!response.ok) throw new Error('Failed to fetch notes');

      const result = await response.json();

      setNotes((prev) => [...prev, ...result.data]);
      setCursor(result.nextCursor);
      setHasMore(result.hasMore);
    } catch (err) {
      setError(err as Error);
    } finally {
      setIsLoading(false);
      loadingRef.current = false;
    }
  }, [cursor, hasMore, limit]);

  // 초기 로드
  useEffect(() => {
    loadMore();
  }, []);

  return {
    notes,
    isLoading,
    error,
    hasMore,
    loadMore,
    refresh: () => {
      setNotes([]);
      setCursor(null);
      setHasMore(true);
      loadMore();
    },
  };
}
```

#### Intersection Observer로 자동 로드

```typescript
// components/notes/infinite-note-list.tsx
'use client';

import { useRef, useEffect } from 'react';
import { useInfiniteNotes } from '@/hooks/use-infinite-notes';
import { NoteCard } from './note-card';
import { Spinner } from '@/components/ui/spinner';

export function InfiniteNoteList() {
  const { notes, isLoading, hasMore, loadMore } = useInfiniteNotes();
  const loadMoreRef = useRef<HTMLDivElement>(null);

  // Intersection Observer로 자동 로드
  useEffect(() => {
    const observer = new IntersectionObserver(
      (entries) => {
        if (entries[0].isIntersecting && hasMore && !isLoading) {
          loadMore();
        }
      },
      { threshold: 0.1 }
    );

    if (loadMoreRef.current) {
      observer.observe(loadMoreRef.current);
    }

    return () => observer.disconnect();
  }, [hasMore, isLoading, loadMore]);

  return (
    <div className="space-y-4">
      {notes.map((note) => (
        <NoteCard key={note.id} note={note} />
      ))}

      {/* 로딩 트리거 */}
      <div ref={loadMoreRef} className="h-20 flex items-center justify-center">
        {isLoading && <Spinner />}
        {!hasMore && notes.length > 0 && (
          <p className="text-muted-foreground">모든 기록을 불러왔습니다</p>
        )}
      </div>
    </div>
  );
}
```

```typescript
import { useMemo } from 'react';

  const stats = useMemo(() => {

    const readingBooks = books.filter(b => b.status === 'reading').length;

    const notesThisWeek = notes.filter(n => {
      const noteDate = new Date(n.created_at);
      const weekAgo = new Date();
      weekAgo.setDate(weekAgo.getDate() - 7);
      return noteDate > weekAgo;
    }).length;

    const notesThisMonth = notes.filter(n => {
      const noteDate = new Date(n.created_at);
      const monthAgo = new Date();
      monthAgo.setMonth(monthAgo.getMonth() - 1);
      return noteDate > monthAgo;
    }).length;

      ? Math.round(notes.length / books.length * 10) / 10
      : 0;

    return {
      completedBooks,
      readingBooks,
      totalNotes: notes.length,
      notesThisWeek,
      notesThisMonth,
      averageNotesPerBook,
    };
  }, [notes, books]);

    <div className="grid grid-cols-2 md:grid-cols-3 gap-4">
      <StatCard title="완독한 책" value={stats.completedBooks} />
      <StatCard title="읽는 중" value={stats.readingBooks} />
      <StatCard title="이번 주 기록" value={stats.notesThisWeek} />
      {/* ... */}
    </div>
```
